
@app.route('/api/stock/technical/<stock_code>', methods=['GET'])
def get_technical_indicators(stock_code):
    """
    获取股票技术指标
    
    Args:
        stock_code: 股票代码
        days: 获取天数，默认60天
    
    Returns:
        JSON格式的技术指标数据，包含MACD、KDJ、均线、ATR等
    """
    try:
        days = int(request.args.get('days', 60))
        clean_code = stock_code.strip().zfill(6)
        
        print(f"[{datetime.now()}] 请求技术指标: {stock_code}, 天数: {days}")
        
        # 获取历史K线数据
        df, method, start_date, end_date = _fetch_history_dataframe_with_fallback(stock_code, months=int(days/30) + 1)
        
        if df is None or df.empty:
            return jsonify({
                'success': False,
                'error': '无法获取K线数据',
                'stockCode': stock_code
            }), 404
        
        # 标准化列名
        column_mapping = {
            '日期': 'date',
            '开盘': 'open',
            '收盘': 'close',
            '最高': 'high',
            '最低': 'low',
            '成交量': 'volume',
            '成交额': 'amount',
            '振幅': 'amplitude',
            '涨跌幅': 'change_pct',
            '涨跌额': 'change_amount',
            '换手率': 'turnover_rate'
        }
        
        df = df.rename(columns=column_mapping)
        
        # 确保必要的列存在
        required_cols = ['date', 'open', 'high', 'low', 'close', 'volume']
        missing_cols = [col for col in required_cols if col not in df.columns]
        if missing_cols:
            return jsonify({
                'success': False,
                'error': f'缺少必要列: {missing_cols}',
                'stockCode': stock_code
            }), 500
        
        # 计算所有技术指标
        df = calculate_all_indicators(df)
        
        # 只返回最近N天的数据
        df = df.tail(days)
        
        # 转换为JSON格式
        df['date'] = pd.to_datetime(df['date']).dt.strftime('%Y-%m-%d')
        
        # 获取最新一天的数据作为摘要
        latest = df.iloc[-1] if not df.empty else {}
        
        result = {
            'stockCode': stock_code,
            'days': len(df),
            'startDate': df.iloc[0]['date'] if not df.empty else None,
            'endDate': df.iloc[-1]['date'] if not df.empty else None,
            'dataSource': method,
            'lastUpdate': datetime.now().isoformat(),
            'latest': {
                'date': latest.get('date'),
                'close': float(latest.get('close', 0)),
                'ma5': float(latest.get('ma5', 0)) if pd.notna(latest.get('ma5')) else None,
                'ma10': float(latest.get('ma10', 0)) if pd.notna(latest.get('ma10')) else None,
                'ma20': float(latest.get('ma20', 0)) if pd.notna(latest.get('ma20')) else None,
                'ma60': float(latest.get('ma60', 0)) if pd.notna(latest.get('ma60')) else None,
                'macd_dif': float(latest.get('macd_dif', 0)) if pd.notna(latest.get('macd_dif')) else None,
                'macd_dea': float(latest.get('macd_dea', 0)) if pd.notna(latest.get('macd_dea')) else None,
                'macd_hist': float(latest.get('macd_hist', 0)) if pd.notna(latest.get('macd_hist')) else None,
                'kdj_k': float(latest.get('kdj_k', 0)) if pd.notna(latest.get('kdj_k')) else None,
                'kdj_d': float(latest.get('kdj_d', 0)) if pd.notna(latest.get('kdj_d')) else None,
                'kdj_j': float(latest.get('kdj_j', 0)) if pd.notna(latest.get('kdj_j')) else None,
                'atr': float(latest.get('atr', 0)) if pd.notna(latest.get('atr')) else None,
                'volume_ma5': float(latest.get('volume_ma5', 0)) if pd.notna(latest.get('volume_ma5')) else None,
            },
            'history': df.to_dict('records')
        }
        
        print(f"[{datetime.now()}] ✅ 成功计算技术指标: {stock_code}, {len(df)}天数据")
        return jsonify({'success': True, 'data': result})
        
    except Exception as e:
        error_msg = str(e)
        error_trace = traceback.format_exc()
        print(f"[{datetime.now()}] ❌ 获取技术指标失败: {error_msg}")
        print(error_trace)
        return jsonify({
            'success': False,
            'error': error_msg,
            'trace': error_trace if os.getenv('FLASK_ENV') == 'development' else None
        }), 500
